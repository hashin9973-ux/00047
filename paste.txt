```rust
//! Declarative block specs. One JSON shape describes a block's UI fields and validation.
//! Used by native blocks (via `Block::spec`) and by user block packs (see custom.rs).
use serde_json::{json, Value};
use std::collections::HashMap;

fn merge(mut base: Value, over: &Value) -> Value {
    if let (Some(b), Some(o)) = (base.as_object_mut(), over.as_object()) {
        for (k, v) in o {
            if k != "preset" {
                b.insert(k.clone(), v.clone());
            }
        }
    }
    base
}

/// Expands `{"preset":"timeout"}` and `{"preset":"onFail"}` into full field definitions.
pub fn expand(f: &Value) -> Value {
    match f.get("preset").and_then(Value::as_str) {
        Some("timeout") => merge(
            json!({"key":"timeoutSec","label":"Timeout","type":"number","min":0,"max":86400,"step":1,"unit":"s","default":0,
                   "help":"0 = no timeout. When it expires the block is marked failed."}),
            f,
        ),
        Some("onFail") => merge(
            json!({"key":"onFail","label":"On failure","type":"select","default":"stop",
                   "options":[["stop","End this branch"],["else","Follow ELSE wires"],["after","Follow AFTER wires"],["abort","Abort automation"]]}),
            f,
        ),
        _ => f.clone(),
    }
}

pub fn fields(spec: &Value) -> Vec<Value> {
    spec.get("fields").and_then(Value::as_array).map(|a| a.iter().map(expand).collect()).unwrap_or_default()
}

fn flag(f: &Value, k: &str) -> bool {
    f.get(k).and_then(Value::as_bool).unwrap_or(false)
}

fn visible(f: &Value, cfg: &Value) -> bool {
    let Some(s) = f.get("show") else { return true };
    let v = cfg.get(s.get("key").and_then(Value::as_str).unwrap_or(""));
    if let Some(eq) = s.get("eq") {
        return v == Some(eq);
    }
    if let Some(ne) = s.get("ne") {
        return v != Some(ne);
    }
    if let Some(list) = s.get("in").and_then(Value::as_array) {
        return v.map(|x| list.contains(x)).unwrap_or(false);
    }
    true
}

fn current<'a>(f: &'a Value, cfg: &'a Value) -> Option<&'a Value> {
    let key = f.get("key")?.as_str()?;
    cfg.get(key).filter(|v| !v.is_null()).or_else(|| f.get("default"))
}

/// "HH:MM" -> minutes since midnight.
pub fn parse_hhmm(s: &str) -> Option<u32> {
    let (h, m) = s.trim().split_once(':')?;
    let (h, m): (u32, u32) = (h.parse().ok()?, m.parse().ok()?);
    if h < 24 && m < 60 {
        Some(h * 60 + m)
    } else {
        None
    }
}

/// Validates a block's saved settings against its spec. Hidden fields are skipped.
pub fn validate(spec: &Value, cfg: &Value) -> Result<(), String> {
    for f in fields(spec) {
        if !visible(&f, cfg) {
            continue;
        }
        let key = f.get("key").and_then(Value::as_str).unwrap_or("");
        let label = f.get("label").and_then(Value::as_str).unwrap_or(key);
        let v = current(&f, cfg);
        match f.get("type").and_then(Value::as_str).unwrap_or("text") {
            "number" => {
                let n = v.and_then(Value::as_f64).filter(|n| n.is_finite()).ok_or_else(|| format!("{label} must be a number"))?;
                if let Some(lo) = f.get("min").and_then(Value::as_f64) {
                    if n < lo {
                        return Err(format!("{label} must be at least {lo}"));
                    }
                }
                if let Some(hi) = f.get("max").and_then(Value::as_f64) {
                    if n > hi {
                        return Err(format!("{label} must be at most {hi}"));
                    }
                }
            }
            "time" => {
                if parse_hhmm(v.and_then(Value::as_str).unwrap_or("")).is_none() {
                    return Err(format!("{label} must be a time like 09:30"));
                }
            }
            "select" => {
                let s = v.and_then(Value::as_str).unwrap_or("");
                if let Some(opts) = f.get("options").and_then(Value::as_array) {
                    if !opts.iter().any(|o| o.get(0).and_then(Value::as_str) == Some(s)) {
                        return Err(format!("{label} has an unknown option"));
                    }
                }
            }
            "bool" => {
                if flag(&f, "required") && v.and_then(Value::as_bool) != Some(true) {
                    return Err(format!("{label} must be ticked"));
                }
            }
            _ => {
                let s = v.and_then(Value::as_str).unwrap_or("").trim();
                if flag(&f, "required") && s.is_empty() {
                    return Err(format!("{label} is required"));
                }
                if let Some(max) = f.get("maxLen").and_then(Value::as_u64) {
                    if s.chars().count() as u64 > max {
                        return Err(format!("{label} must be {max} characters or fewer"));
                    }
                }
            }
        }
    }
    Ok(())
}

/// Every field's current value (saved or default) as a string. Used for command templates and env vars.
pub fn values(spec: &Value, cfg: &Value) -> HashMap<String, String> {
    let mut m = HashMap::new();
    for f in fields(spec) {
        let Some(key) = f.get("key").and_then(Value::as_str) else { continue };
        let s = match current(&f, cfg) {
            Some(Value::String(s)) => s.clone(),
            Some(Value::Bool(b)) => b.to_string(),
            Some(Value::Number(n)) => n.to_string(),
            _ => String::new(),
        };
        m.insert(key.to_string(), s);
    }
    m
}
```

```rust
//! PowerShell script. No shell string building: arguments are passed separately, and PowerShell scripts read
//! field values from environment variables (FIELD_<KEY>), so field text can never be injected as code.

static DIR: OnceLock<PathBuf> = OnceLock::new();
static ERRORS: Mutex<Vec<String>> = Mutex::new(Vec::new());

// ---------- templates: "{field}" substitution, "{{" and "}}" are literal braces ----------
fn placeholders(t: &str) -> Vec<String> {
    let b: Vec<char> = t.chars().collect();
    let (mut out, mut i) = (vec![], 0);
    while i < b.len() {
        if b[i] == '{' {
            if i + 1 < b.len() && b[i + 1] == '{' {
                i += 2;
                continue;
            }
            if let Some(j) = b[i + 1..].iter().position(|&c| c == '}') {
                out.push(b[i + 1..i + 1 + j].iter().collect());
                i += j + 2;
                continue;
            }
        }
        i += 1;
    }
    out
}

fn subst(t: &str, vals: &HashMap<String, String>) -> String {
    let b: Vec<char> = t.chars().collect();
    let (mut out, mut i) = (String::new(), 0);
    while i < b.len() {
        if b[i] == '{' {
            if i + 1 < b.len() && b[i + 1] == '{' {
                out.push('{');
                i += 2;
                continue;
            }
            if let Some(j) = b[i + 1..].iter().position(|&c| c == '}') {
                let k: String = b[i + 1..i + 1 + j].iter().collect();
                out.push_str(vals.get(&k).map(String::as_str).unwrap_or(""));
                i += j + 2;
                continue;
            }
        }
        if b[i] == '}' && i + 1 < b.len() && b[i + 1] == '}' {
            out.push('}');
            i += 2;
            continue;
        }
        out.push(b[i]);
        i += 1;
    }
    out
}

// ---------- spec checking ----------
fn check_spec(s: &Value) -> Result<String, String> {
    let kind = s.get("kind").and_then(Value::as_str).ok_or("a block has no \"kind\"")?;
    if !kind.starts_with("user.")
        || kind.len() > 60
        || !kind.chars().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '.' || c == '_')
    {
        return Err(format!("kind '{kind}' must start with \"user.\" and use only a-z, 0-9, '.' and '_'"));
    }
    match s.get("category").and_then(Value::as_str) {
        Some("condition") | Some("action") => {}
        _ => return Err(format!("{kind}: category must be \"condition\" or \"action\"")),
    }
    let title = s.get("title").and_then(Value::as_str).unwrap_or("").trim();
    if title.is_empty() || title.chars().count() > 40 {
        return Err(format!("{kind}: title is required (max 40 characters)"));
    }
    let mut keys: Vec<String> = vec![];
    for f in spec::fields(s) {
        let key = f.get("key").and_then(Value::as_str).ok_or_else(|| format!("{kind}: a field has no key"))?;
        if key.is_empty() || !key.chars().all(|c| c.is_ascii_alphanumeric() || c == '_') {
            return Err(format!("{kind}: field key '{key}' may only use letters, digits and _"));
        }
        if keys.iter().any(|k| k == key) {
            return Err(format!("{kind}: duplicate field key '{key}'"));
        }
        match f.get("type").and_then(Value::as_str).unwrap_or("text") {
            "number" => {
                if f.get("min").and_then(Value::as_f64).is_none() || f.get("max").and_then(Value::as_f64).is_none() {
                    return Err(format!("{kind}: number field '{key}' needs numeric min and max"));
                }
            }
            "select" => {
                let ok = f.get("options").and_then(Value::as_array).map(|o| !o.is_empty() && o.iter().all(|x| x.get(0).and_then(Value::as_str).is_some() && x.get(1).and_then(Value::as_str).is_some())).unwrap_or(false);
                if !ok {
                    return Err(format!("{kind}: select field '{key}' needs options like [[\"value\",\"Label\"]]"));
                }
            }
            "text" | "file" | "bool" | "time" => {}
            other => return Err(format!("{kind}: field '{key}' has unknown type '{other}'")),
        }
        keys.push(key.to_string());
    }
    let run = s.get("run").ok_or_else(|| format!("{kind}: missing \"run\""))?;
    let mut texts: Vec<String> = vec![];
    match run.get("type").and_then(Value::as_str) {
        Some("command") => {
            let p = run.get("program").and_then(Value::as_str).unwrap_or("");
            if p.trim().is_empty() {
                return Err(format!("{kind}: run.program is required"));
            }
            texts.push(p.to_string());
            if let Some(a) = run.get("args") {
                for x in a.as_array().ok_or_else(|| format!("{kind}: run.args must be an array of strings"))? {
                    texts.push(x.as_str().ok_or_else(|| format!("{kind}: run.args must be an array of strings"))?.to_string());
                }
            }
            if let Some(c) = run.get("cwd").and_then(Value::as_str) {
                texts.push(c.to_string());
            }
        }
        Some("powershell") => {
            if run.get("script").and_then(Value::as_str).unwrap_or("").trim().is_empty() {
                return Err(format!("{kind}: run.script is required"));
            }
        }
        _ => return Err(format!("{kind}: run.type must be \"command\" or \"powershell\"")),
    }
    for t in texts {
        for p in placeholders(&t) {
            if !keys.contains(&p) {
                return Err(format!("{kind}: placeholder {{{p}}} is not a field of this block"));
            }
        }
    }
    Ok(kind.to_string())
}

// ---------- the block ----------
struct CustomBlock {
    spec: Value,
    title: &'static str,
    cat: Cat,
    timeout: bool,
}

impl CustomBlock {
    fn new(spec: Value) -> Self {
        let title: &'static str = Box::leak(spec["title"].as_str().unwrap_or("Custom").trim().to_string().into_boxed_str());
        let cat = if spec["category"] == "condition" { Cat::Condition } else { Cat::Action };
        let timeout = spec::fields(&spec).iter().any(|f| f["key"] == "timeoutSec");
        CustomBlock { spec, title, cat, timeout }
    }
}

fn tail(b: &[u8]) -> String {
    let s = String::from_utf8_lossy(b);
    let s = s.trim();
    let n = s.chars().count();
    if n > 400 {
        format!("…{}", s.chars().skip(n - 400).collect::<String>())
    } else {
        s.to_string()
    }
}

#[async_trait]
impl Block for CustomBlock {
    fn title(&self) -> &'static str {
        self.title
    }
    fn category(&self) -> Cat {
        self.cat
    }
    fn supports_timeout(&self) -> bool {
        self.timeout
    }
    fn risky(&self) -> bool {
        self.cat == Cat::Action
    }
    fn spec(&self) -> Option<Value> {
        Some(self.spec.clone())
    }
    fn validate(&self, cfg: &Value) -> Result<(), String> {
        spec::validate(&self.spec, cfg)
    }
    async fn run(&self, cx: &NodeCx) -> Result<Outcome, String> {
        let run = &self.spec["run"];
        let vals = spec::values(&self.spec, &cx.cfg);
        let (program, args): (String, Vec<String>) = if run["type"] == "powershell" {
            (
                "powershell.exe".into(),
                vec!["-NoProfile".into(), "-NonInteractive".into(), "-ExecutionPolicy".into(), "Bypass".into(), "-Command".into(), run["script"].as_str().unwrap_or("").to_string()],
            )
        } else {
            let skip_empty = run["skipEmptyArgs"].as_bool().unwrap_or(false);
            let args = run["args"]
                .as_array()
                .map(|a| a.iter().filter_map(Value::as_str).map(|t| subst(t, &vals)).filter(|s| !(skip_empty && s.is_empty())).collect())
                .unwrap_or_default();
            (subst(run["program"].as_str().unwrap_or(""), &vals), args)
        };
        if program.trim().is_empty() {
            return Err("the program to run is empty (check this block's settings)".into());
        }
        let mut cmd = tokio::process::Command::new(&program);
        cmd.args(&args).stdin(Stdio::null()).stdout(Stdio::piped()).stderr(Stdio::piped()).kill_on_drop(true);
        for (k, v) in &vals {
            cmd.env(format!("FIELD_{}", k.to_uppercase()), v);
        }
        if let Some(cwd) = run["cwd"].as_str() {
            let cwd = subst(cwd, &vals);
            if !cwd.trim().is_empty() {
                cmd.current_dir(cwd);
            }
        }
        #[cfg(windows)]
        cmd.creation_flags(0x0800_0000); // CREATE_NO_WINDOW

        (cx.report)("running", None, format!("Running {}", self.title));
        if !run["wait"].as_bool().unwrap_or(true) {
            cmd.stdout(Stdio::null()).stderr(Stdio::null()).kill_on_drop(false);
            cmd.spawn().map_err(|e| format!("could not start '{program}': {e}"))?;
            (cx.log)("info", format!("started {program} (not waiting for it)"));
            return Ok(Outcome::Satisfied);
        }
        let child = cmd.spawn().map_err(|e| format!("could not start '{program}': {e}"))?;
        let out = tokio::select! {
            r = child.wait_with_output() => r.map_err(|e| e.to_string())?,
            _ = cx.token.cancelled() => return Err("cancelled".into()),
        };
        let code = out.status.code().unwrap_or(-1) as i64;
        let ok_codes: Vec<i64> = run["okExitCodes"].as_array().map(|a| a.iter().filter_map(Value::as_i64).collect()).unwrap_or_else(|| vec![0]);
        let (stdout, stderr) = (tail(&out.stdout), tail(&out.stderr));
        if run["capture"].as_bool().unwrap_or(true) && !stdout.is_empty() {
            (cx.log)("info", format!("output: {stdout}"));
        }
        let ok = ok_codes.contains(&code);
        if self.cat == Cat::Condition {
            return Ok(if ok { Outcome::Satisfied } else { Outcome::NotSatisfied });
        }
        if ok {
            Ok(Outcome::Satisfied)
        } else {
            Err(format!("exited with code {code}{}", if stderr.is_empty() { String::new() } else { format!(": {stderr}") }))
        }
    }
}

// ---------- loading ----------
fn parse_pack(path: &Path) -> Result<Vec<Value>, String> {
    if path.extension().map(|x| x != "json").unwrap_or(true) {
        return Err("a block pack must be a .json file".into());
    }
    let meta = std::fs::metadata(path).map_err(|e| e.to_string())?;
    if meta.len() > 1_000_000 {
        return Err("the pack is larger than 1 MB".into());
    }
    let text = std::fs::read_to_string(path).map_err(|e| e.to_string())?;
    let v: Value = serde_json::from_str(text.trim_start_matches('\u{feff}')).map_err(|e| format!("invalid JSON: {e}"))?;
    Ok(if let Some(a) = v.get("blocks").and_then(Value::as_array) {
        a.clone()
    } else if let Value::Array(a) = &v {
        a.clone()
    } else {
        vec![v]
    })
}

/// Called once while the block registry is built.
pub fn load_into(m: &mut HashMap<&'static str, Box<dyn Block>>) {
    let Some(dir) = DIR.get() else { return };
    let Ok(rd) = std::fs::read_dir(dir) else { return };
    let mut files: Vec<PathBuf> = rd.filter_map(|e| e.ok()).map(|e| e.path()).filter(|p| p.extension().map(|x| x == "json").unwrap_or(false)).collect();
    files.sort();
    for p in files {
        let name = p.file_name().map(|n| n.to_string_lossy().to_string()).unwrap_or_default();
        let specs = match parse_pack(&p) {
            Ok(s) => s,
            Err(e) => {
                push_err(format!("{name}: {e}"));
                continue;
            }
        };
        for s in specs {
            match check_spec(&s) {
                Ok(kind) => {
                    if m.contains_key(kind.as_str()) {
                        push_err(format!("{name}: kind '{kind}' is already defined"));
                        continue;
                    }
                    let leaked: &'static str = Box::leak(kind.into_boxed_str());
                    m.insert(leaked, Box::new(CustomBlock::new(s)));
                }
                Err(e) => push_err(format!("{name}: {e}")),
            }
        }
    }
}

/// Specs of every block that provides one (native spec-driven blocks + user blocks) for the UI.
pub fn all_specs() -> Value {
    let mut specs: Vec<Value> = vec![];
    for (kind, b) in registry().iter() {
        if let Some(mut s) = b.spec() {
            s["kind"] = json!(kind);
            s["origin"] = json!(if kind.starts_with("user.") { "custom" } else { "native" });
            specs.push(s);
        }
    }
    specs.sort_by(|a, b| a["title"].as_str().cmp(&b["title"].as_str()));
    json!({
        "specs": specs,
        "errors": load_errors(),
        "dir": blocks_dir().map(|d| d.to_string_lossy().to_string()).unwrap_or_default(),
    })
}

#[derive(Serialize)]
pub struct BlockPreview {
    pub kind: String,
    pub title: String,
    pub category: String,
    pub command: String,
}
#[derive(Serialize)]
pub struct PackPreview {
    pub file: String,
    pub path: String,
    pub blocks: Vec<BlockPreview>,
    pub errors: Vec<String>,
}

fn describe(s: &Value) -> String {
    let run = &s["run"];
    if run["type"] == "powershell" {
        return format!("powershell -Command {}", run["script"].as_str().unwrap_or(""));
    }
    let mut parts = vec![run["program"].as_str().unwrap_or("").to_string()];
    if let Some(a) = run["args"].as_array() {
        parts.extend(a.iter().map(|x| format!("\"{}\"", x.as_str().unwrap_or(""))));
    }
    parts.join(" ")
}

pub fn preview(path: &Path) -> Result<PackPreview, String> {
    let specs = parse_pack(path)?;
    let (mut blocks, mut errors) = (vec![], vec![]);
    for s in &specs {
        match check_spec(s) {
            Ok(kind) => blocks.push(BlockPreview {
                kind,
                title: s["title"].as_str().unwrap_or("").to_string(),
                category: s["category"].as_str().unwrap_or("").to_string(),
                command: describe(s),
            }),
            Err(e) => errors.push(e),
        }
    }
    if blocks.is_empty() {
        return Err(format!("no valid blocks in this file{}", if errors.is_empty() { String::new() } else { format!(": {}", errors.join("; ")) }));
    }
    let file = path.file_name().map(|n| n.to_string_lossy().to_string()).unwrap_or_default();
    Ok(PackPreview { file, path: path.to_string_lossy().to_string(), blocks, errors })
}

pub fn install(path: &Path) -> Result<String, String> {
    let pv = preview(path)?;
    let dir = DIR.get().ok_or("blocks folder is not initialised")?;
    std::fs::create_dir_all(dir).map_err(|e| e.to_string())?;
    let stem: String = path
        .file_stem()
        .map(|s| s.to_string_lossy().to_string())
        .unwrap_or_else(|| "pack".into())
        .chars()
        .map(|c| if c.is_ascii_alphanumeric() || c == '-' || c == '_' { c } else { '_' })
        .collect();
    let dest = dir.join(format!("{stem}.json"));
    if dest.exists() && std::fs::canonicalize(&dest).ok() == std::fs::canonicalize(path).ok() {
        return Ok(pv.file);
    }
    std::fs::copy(path, &dest).map_err(|e| format!("copy: {e}"))?;
    Ok(pv.file)
```

```rust
        ],

struct Schedule;
fn in_window(c: &Value) -> Result<bool, String> {
    let get = |k: &str, d: &str| {
        let t = text(c, k);
        if t.is_empty() { d.to_string() } else { t }
    };
    let s = spec::parse_hhmm(&get("start", "09:00")).ok_or("Start time must look like 09:00")?;
    let e = spec::parse_hhmm(&get("end", "17:00")).ok_or("End time must look like 17:00")?;
    let (dow, now) = sys::local_weekday_minutes();
    let day_ok = match text(c, "days").as_str() {
        "weekdays" => (1..=5).contains(&dow),
        "weekends" => dow == 0 || dow == 6,
        _ => true,
    };
    let time_ok = s == e || if s < e { now >= s && now < e } else { now >= s || now < e };
    Ok(day_ok && time_ok)
}

        let wait = text(&cx.cfg, "mode") == "wait";
```

```js
const PRESETS = { timeout: TIMEOUT, onFail: ONFAIL };
const showFn = s => {
  if (!s || typeof s !== 'object') return undefined;
  return c => {
    const v = c[s.key];
    if ('eq' in s) return v === s.eq;
    if ('ne' in s) return v !== s.ne;
    if (Array.isArray(s.in)) return s.in.includes(v);
    return true;
  };

export function compileSpec(sp) {
  const cat = sp.category === 'condition' ? 'cond' : sp.category === 'target' ? 'tgt' : 'act';
  const fields = (sp.fields || []).map(f => {
    const o = { ...(f.preset ? PRESETS[f.preset] : {}), ...f };
    delete o.preset;
    if (o.show && typeof o.show === 'object') o.show = showFn(o.show);
    return o;
  });
  const defaults = {};
  for (const f of fields) {
    defaults[f.key] = f.default !== undefined ? f.default
      : f.type === 'bool' ? false
      : f.type === 'number' ? (f.min ?? 0)
      : f.type === 'select' ? f.options[0][0] : '';
  }
  Object.assign(defaults, sp.defaults || {});

    const o = { ...c };

    return o;
  };
  return {

    defaults, fields, summary: c => tpl(sp.summary || sp.title, labelled(c)), help: sp.help, origin: sp.origin,
  };
}
```

```js
    const minX = Math.min(...c.nodes.map(n => n.x)), minY = Math.min(...c.nodes.map(n => n.y));
    const bx = at ? at.x - minX : 40, by = at ? at.y - minY : 40;
    const fits = (ox, oy) => !c.nodes.some(n => {
      const r = { x: n.x + ox, y: n.y + oy, w: W, h: n._h || 90 };
      return this.auto.nodes.some(q => this.collide(r, this.rectOf(q)));
    });
    let ox = Math.round(bx), oy = Math.round(by);
    for (const [dx, dy] of OFFS) { if (fits(bx + dx, by + dy)) { ox = Math.round(bx + dx); oy = Math.round(by + dy); break; } }
    const map = new Map(), created = [];
    for (const n of c.nodes) {
      const m = structuredClone(n); delete m._h;
      m.id = uid(); m.x = n.x + ox; m.y = n.y + oy;
      map.set(n.id, m.id); created.push(m.id);
      this.auto.nodes.push(m); this.addEl(m);
    }
    for (const e of c.edges) this.auto.edges.push({ ...structuredClone(e), id: uid(), from: map.get(e.from), to: map.get(e.to) });
    if (!at) c.nodes.forEach(n => { n.x += ox; n.y += oy; });
    this.select(created); this.refreshAll(); this.scheduleWires(); this.changed();
  }
```

**Create `00047/docs/ADDING_BLOCKS.md` (new file; one page for users and developers).**

````markdown
# Adding blocks to 00047

A block is a Condition (IF/ELSE) or an Action. Pick the route that fits.

## 1. Users: a block pack (JSON, no coding)

Put a `.json` file in the blocks folder (Settings → Custom blocks → Open folder), then choose **Reload**
(00047 restarts). Or use **Install pack…**, which shows exactly what each block will run before copying it.
An example is in `docs/examples/blocks/sample-pack.json`.

```json
{
  "kind": "user.open_url",            // must start with "user."; a-z 0-9 . _
  "category": "action",               // "condition" or "action"
  "title": "Open URL",
  "icon": "play",                     // clock chip trim hourglass play repeat window file note proc stop bell
  "description": "Opens a web address.",
  "fields": [
    { "key": "url", "label": "Address", "type": "text", "required": true, "maxLen": 500 },
    { "preset": "timeout" },          // adds a Timeout setting (and enables the timeout)
    { "preset": "onFail" }            // adds the "On failure" setting
  ],
  "summary": "Open {url}",
  "run": { "type": "powershell", "script": "Start-Process $env:FIELD_URL" }
}
```

A file may hold one block, an array of blocks, or `{ "blocks": [ ... ] }`.

**Run types**
- `command`: `{ "type":"command", "program":"C:\\Tools\\x.exe", "args":["--in","{file}"], "cwd":"{folder}", "skipEmptyArgs":true, "wait":true, "okExitCodes":[0], "capture":true }`
  Arguments are passed one by one (no shell), so field text cannot inject extra commands.
- `powershell`: `{ "type":"powershell", "script":"..." }`. Field values arrive as environment variables
  `FIELD_<KEY>` (key upper-cased), for example `$env:FIELD_URL`. They are never pasted into the script text.

**Results.** Exit code in `okExitCodes` (default `[0]`): a Condition is satisfied (else not satisfied); an Action
succeeds (else it fails and follows its "On failure" setting). `wait:false` starts the program and continues at once.
`{field}` placeholders work in `program`, `args` and `cwd`; use `{{` and `}}` for literal braces.

**Fields.** `type`: `text`, `file` (adds Browse), `number` (needs `min` and `max`), `select` (`options`: `[["value","Label"]]`),
`bool`, `time` (HH:MM). Common keys: `key`, `label`, `default`, `required`, `maxLen`, `unit`, `step`, `help`,
`show` (`{"key":"mode","eq":"wait"}`, `ne`, or `in:[...]`).

**Safety.** A pack runs programs with your permissions. Install only packs you trust. Actions from packs fall under
the runaway guard (at most 12 runs per 10 s). Imported automations cannot install packs.

## 2. Developers: a native block (Rust)

1. In `src-tauri/src/blocks.rs` add a struct and `impl Block for ...`:
   - `title()`, `category()`: required.
   - `spec()`: return the same JSON as above (without `run`) and the UI is generated automatically:
     **no change to `ui/blocks.js` is needed**.
   - `validate(cfg)`: `spec::validate(&my_spec(), cfg)` checks every field from the spec for you.
   - `run(cx)`: do the work and return `Outcome::Satisfied` / `NotSatisfied`, or `Err("reason")`.
     Use `cx.report(state, progress, text)`, `cx.log(level, text)`, `cx.resolve()` (wired targets),
     `cx.token` (cancellation), `cx.visit` (times reached).
   - `supports_timeout()` true if the spec has a `timeout` preset field; `risky()` true for side-effect actions.
2. Register it in `registry()`: `m.insert("cond.my_block", Box::new(MyBlock));`
3. `Time window` (`cond.schedule`) is the reference example.

Blocks that need a custom editor widget can still be defined by hand in `ui/blocks.js` as before; a hand-written
definition wins over a spec with the same kind.
````
