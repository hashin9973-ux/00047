struct ProcRunning;
#[async_trait]
impl Block for ProcRunning {
    fn title(&self) -> &'static str {
        "Process Running"
    }
    fn category(&self) -> Cat {
        Cat::Condition
    }
    fn supports_timeout(&self) -> bool {
        true
    }
    fn validate(&self, c: &Value) -> Result<(), String> {
        if text(c, "name").is_empty() {
            return Err("Process name is required".into());
        }
        let mode = text(c, "mode");
        if !matches!(mode.as_str(), "" | "check" | "wait_start" | "wait_stop") {
            return Err("Unknown mode".into());
        }
        if mode.starts_with("wait") {
            rng(c, "pollSec", 2.0, 0.25, 3600.0, "Poll interval")?;
            check_timeout(c)?;
        }
        Ok(())
    }
    async fn run(&self, cx: &NodeCx) -> Result<Outcome, String> {
        let name = text(&cx.cfg, "name");
        let mode = text(&cx.cfg, "mode");
        let poll = rng(&cx.cfg, "pollSec", 2.0, 0.25, 3600.0, "Poll interval")?;
        let wait_start = mode == "wait_start";
        let wait_stop = mode == "wait_stop";
        loop {
            let running = sys::find_pid_by_name(&name).is_ok();
            (cx.report)(
                if wait_start || wait_stop { "waiting" } else { "running" },
                None,
                format!("{name}: {}", if running { "running" } else { "not running" }),
            );
            if wait_start {
                if running {
                    return Ok(Outcome::Satisfied);
                }
            } else if wait_stop {
                if !running {
                    return Ok(Outcome::Satisfied);
                }
            } else {
                return Ok(if running { Outcome::Satisfied } else { Outcome::NotSatisfied });
            }
            tokio::select! {
                _ = tokio::time::sleep(Duration::from_secs_f64(poll)) => {},
                _ = cx.token.cancelled() => return Err("cancelled".into()),
            }
        }
    }
}

struct StopProc;
#[async_trait]
impl Block for StopProc {
    fn title(&self) -> &'static str {
        "Stop Process"
    }
    fn category(&self) -> Cat {
        Cat::Action
    }
    fn supports_timeout(&self) -> bool {
        true
    }
    fn validate(&self, c: &Value) -> Result<(), String> {
        if text(c, "mode") != "target" && text(c, "name").is_empty() {
            return Err("Process name is required".into());
        }
        check_timeout(c)
    }
    async fn run(&self, cx: &NodeCx) -> Result<Outcome, String> {
        let force = flag(&cx.cfg, "force", false);
        let (pid, name) = if text(&cx.cfg, "mode") == "target" {
            first_process(cx)?
        } else {
            let n = text(&cx.cfg, "name");
            (sys::find_pid_by_name(&n)?, n)
        };
        if pid == sys::own_pid() {
            return Err("refusing to stop 00047 itself".into());
        }
        (cx.report)("running", None, format!("Stopping {name} ({pid})"));
        let msg = tokio::task::spawn_blocking(move || sys::stop_process(pid, force))
            .await
            .map_err(|e| e.to_string())??;
        (cx.log)("info", format!("{name} ({pid}): {msg}"));
        Ok(Outcome::Satisfied)
    }
}

/// Closes a process: WM_CLOSE to its windows, waits 5 s, then TerminateProcess (or straight to terminate if force).
#[cfg(windows)]
pub fn stop_process(pid: u32, force: bool) -> Result<String, String> {
    use windows::Win32::Foundation::{CloseHandle, BOOL, HWND, LPARAM, WAIT_OBJECT_0, WPARAM};
    use windows::Win32::System::Threading::{OpenProcess, TerminateProcess, WaitForSingleObject, PROCESS_SYNCHRONIZE, PROCESS_TERMINATE};
    use windows::Win32::UI::WindowsAndMessaging::{EnumWindows, GetWindowThreadProcessId, PostMessageW, WM_CLOSE};

    unsafe extern "system" fn close_cb(hwnd: HWND, lp: LPARAM) -> BOOL {
        let mut p = 0u32;
        GetWindowThreadProcessId(hwnd, Some(&mut p));
        if p == lp.0 as u32 {
            let _ = PostMessageW(hwnd, WM_CLOSE, WPARAM(0), LPARAM(0));
        }
        BOOL(1)
    }

    let h = unsafe { OpenProcess(PROCESS_SYNCHRONIZE | PROCESS_TERMINATE, false, pid) }
        .map_err(|e| format!("open process {pid}: {e}"))?;
    let res = (|| -> Result<String, String> {
        if !force {
            unsafe {
                let _ = EnumWindows(Some(close_cb), LPARAM(pid as isize));
            }
            if unsafe { WaitForSingleObject(h, 5000) } == WAIT_OBJECT_0 {
                return Ok("closed gracefully".to_string());
            }
        }
        unsafe { TerminateProcess(h, 1) }.map_err(|e| format!("TerminateProcess: {e}"))?;
        unsafe {
            WaitForSingleObject(h, 3000);
        }
        let m = if force { "force-killed" } else { "did not close within 5 s; force-killed" };
        Ok(m.to_string())
    })();
    unsafe {
        let _ = CloseHandle(h);
    }
    res
}

/// Toast notification. Uses the built-in PowerShell AppUserModelID because an unpackaged app has no registered one.
#[cfg(windows)]
pub fn show_notification(title: &str, body: &str, sound: bool) -> Result<(), String> {
    use windows::core::HSTRING;
    use windows::Data::Xml::Dom::XmlDocument;
    use windows::UI::Notifications::{ToastNotification, ToastNotificationManager};

    unsafe {
        let _ = windows::Win32::System::WinRT::RoInitialize(windows::Win32::System::WinRT::RO_INIT_MULTITHREADED);
    }
    let esc = |s: &str| s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;").replace('\'', "&apos;");
    let audio = if sound { "" } else { "<audio silent=\"true\"/>" };
    let xml = format!(
        "<toast><visual><binding template=\"ToastGeneric\"><text>{}</text><text>{}</text></binding></visual>{}</toast>",
        esc(title),
        esc(body),
        audio
    );
    let doc = XmlDocument::new().map_err(|e| format!("XmlDocument: {e}"))?;
    doc.LoadXml(&HSTRING::from(xml)).map_err(|e| format!("LoadXml: {e}"))?;
    let toast = ToastNotification::CreateToastNotification(&doc).map_err(|e| format!("toast: {e}"))?;
    let aumid = HSTRING::from(r"{1AC14E77-02E7-4E5D-B744-2EB1AE5198B7}\WindowsPowerShell\v1.0\powershell.exe");
    ToastNotificationManager::CreateToastNotifierWithId(&aumid)
        .map_err(|e| format!("notifier: {e}"))?
        .Show(&toast)
        .map_err(|e| format!("show: {e}"))
}
